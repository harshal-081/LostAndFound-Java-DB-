Core Tables

CREATE TABLE reports (
  report_id INT PRIMARY KEY AUTO_INCREMENT,
  name VARCHAR(100) NOT NULL,
  age INT NOT NULL CHECK (age BETWEEN 1 AND 120),
  gender VARCHAR(20) NOT NULL,
  address VARCHAR(255) NOT NULL,
  phone_no BIGINT NOT NULL,
  report_date DATE NOT NULL,
  location VARCHAR(150) NOT NULL,
  description VARCHAR(500) NOT NULL,
  status VARCHAR(30) DEFAULT 'Missing',
  reporter_name VARCHAR(100) NOT NULL,
  contact_no VARCHAR(15) NOT NULL,
  relationship VARCHAR(50),
  status_remarks VARCHAR(255)
);

Frontend Actions

-- Add report from frontend POST /api/reports
CALL sp_add_report('Aarav Sharma', 16, 'Male', 'Andheri West, Mumbai',
  9876543210, '2026-04-10', 'Andheri Railway Station',
  'Blue jeans and white shirt', 'Neha Sharma', '9876543210',
  'Parent', 'Missing', @new_id);

-- Update status from frontend PUT /api/reports/{id}/status
CALL sp_update_report_status(@new_id, 'Found', 'Found near station help desk');

-- Delete from frontend DELETE /api/reports/{id}
CALL sp_delete_report(@new_id);

-- Function used by triggers

-- Trigger flow
trg_reports_before_insert  -> normalizes status and contact number
trg_reports_after_insert   -> writes report_status_history and report_audit_log
trg_reports_before_update  -> normalizes changed status and default remarks
trg_reports_after_update   -> stores every status change automatically
trg_reports_after_delete   -> keeps a delete audit record

Java JDBC Code Sample

// 1. JDBC Connection
public class DBConnection {
    static final String URL = "jdbc:mysql://localhost:3306/lost_found_db";
    static final String USER = "root";
    static final String PASS = "your_password";

    public static Connection getConnection() throws SQLException {
        return DriverManager.getConnection(URL, USER, PASS);
    }
}

// 2. Insert Report
public void insertReport(String name, int age) {
    String sql = "{CALL sp_add_report(?, ?, ?, ..., ?)}";
    try (Connection con = DBConnection.getConnection();
         CallableStatement cs = con.prepareCall(sql)) {
        cs.setString(1, name);
        cs.setInt(2, age);
        cs.execute();
    } catch (SQLException e) { e.printStackTrace(); }
}

// 3. Update Status
public void updateStatus(int id, String status) {
    String sql = "{CALL sp_update_report_status(?, ?, ?)}";
    try (Connection con = DBConnection.getConnection();
         CallableStatement cs = con.prepareCall(sql)) {
        cs.setInt(1, id);
        cs.setString(2, status);
        cs.setString(3, "Updated from frontend");
        cs.execute();
    } catch (SQLException e) { e.printStackTrace(); }
}
